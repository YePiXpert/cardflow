## 0. 检查脚本原文（/tmp/t003/tools/ast_equiv.py，一次性工具，不入库）
```python
"""Compare PoC modules with their migrated copies at the AST level (T-003 evidence).

Normalisation applied to BOTH sides before comparing:
  * type annotations removed (function arguments/returns; annotated assignments keep their
    target and value, the annotation becomes the placeholder ``_``),
  * docstrings whitespace-normalised (same words, any line wrapping),
  * import statements removed from the code body (they are listed and diffed separately,
    after resolving the PoC's relative imports and renaming ``cardflow_ref`` to
    ``cardflow.planner.engine``),
  * PEP 695 ``type X = ...`` aliases removed (typing-only, lazily evaluated).
Comments, ``# noqa`` markers and formatting do not exist in the AST, so they never show up.
Any remaining difference is printed as a unified diff of ``ast.unparse`` output.

usage: python ast_equiv.py OLD NEW [OLD NEW ...]
"""

import ast
import difflib
import sys
from pathlib import Path


class _Normalise(ast.NodeTransformer):
    def _strip_args(self, args: ast.arguments) -> None:
        for a in [*args.posonlyargs, *args.args, *args.kwonlyargs]:
            a.annotation = None
        if args.vararg:
            args.vararg.annotation = None
        if args.kwarg:
            args.kwarg.annotation = None

    def _docstring(self, node: ast.AST) -> None:
        body = getattr(node, "body", None)
        if (
            body
            and isinstance(body[0], ast.Expr)
            and isinstance(body[0].value, ast.Constant)
            and isinstance(body[0].value.value, str)
        ):
            body[0].value.value = " ".join(body[0].value.value.split())

    def visit_Module(self, node: ast.Module) -> ast.AST:
        self._docstring(node)
        return self.generic_visit(node)

    def visit_ClassDef(self, node: ast.ClassDef) -> ast.AST:
        self._docstring(node)
        return self.generic_visit(node)

    def visit_FunctionDef(self, node: ast.FunctionDef) -> ast.AST:
        self._docstring(node)
        node.returns = None
        self._strip_args(node.args)
        return self.generic_visit(node)

    def visit_Lambda(self, node: ast.Lambda) -> ast.AST:
        return self.generic_visit(node)

    def visit_AnnAssign(self, node: ast.AnnAssign) -> ast.AST:
        node.annotation = ast.Name("_", ast.Load())
        return self.generic_visit(node)

    def visit_TypeAlias(self, node: ast.AST) -> None:
        return None  # PEP 695 `type X = ...`: lazily evaluated, typing-only

    def visit_Import(self, node: ast.Import) -> None:
        return None

    def visit_ImportFrom(self, node: ast.ImportFrom) -> None:
        return None


OLD_PKG, NEW_PKG = "cardflow_ref", "cardflow.planner.engine"


def _imports(tree: ast.Module) -> list[str]:
    """Imported names; relative imports resolved inside the PoC package, PoC package renamed."""
    out = []
    for node in ast.walk(tree):
        if isinstance(node, ast.Import):
            out += [f"import {a.name}" + (f" as {a.asname}" if a.asname else "") for a in node.names]
        elif isinstance(node, ast.ImportFrom):
            if node.level:  # only the PoC package uses relative imports (level 1)
                mod = OLD_PKG + (f".{node.module}" if node.module else "")
            else:
                mod = node.module or ""
            if mod == OLD_PKG or mod.startswith(OLD_PKG + "."):
                mod = NEW_PKG + mod[len(OLD_PKG) :]
            out += [
                f"from {mod} import {a.name}" + (f" as {a.asname}" if a.asname else "")
                for a in node.names
            ]
    return sorted(out)


def _normalised(path: Path) -> tuple[str, list[str]]:
    tree = ast.parse(path.read_text(encoding="utf-8"), filename=str(path))
    imports = _imports(tree)
    tree = ast.fix_missing_locations(_Normalise().visit(tree))
    return ast.unparse(tree), imports


def main(argv: list[str]) -> int:
    pairs = list(zip(argv[0::2], argv[1::2], strict=True))
    differing = 0
    for old, new in pairs:
        old_code, old_imports = _normalised(Path(old))
        new_code, new_imports = _normalised(Path(new))
        print(f"== {old}\n   {new}")
        code_diff = list(
            difflib.unified_diff(
                old_code.splitlines(), new_code.splitlines(), "PoC", "migrated", lineterm="", n=1
            )
        )
        if code_diff:
            differing += 1
            print("   code: DIFFERS (normalised unified diff below)")
            print("\n".join("     " + line for line in code_diff))
        else:
            print("   code: IDENTICAL after normalisation")
        removed = [i for i in old_imports if i not in new_imports]
        added = [i for i in new_imports if i not in old_imports]
        if removed or added:
            print("   imports:")
            for i in removed:
                print(f"     - {i}")
            for i in added:
                print(f"     + {i}")
        else:
            print("   imports: identical")
    print(f"\n{len(pairs)} file pairs, {differing} with code differences after normalisation")
    return 0


if __name__ == "__main__":
    raise SystemExit(main(sys.argv[1:]))
```
