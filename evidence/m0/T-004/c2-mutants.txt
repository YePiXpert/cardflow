```python
"""R3E-16 mutation check (T-004c): the six mutants that survived the wave-3 review, re-applied to
the model 1.1.0 engine. Each mutant = one textual replacement in one engine file, applied to an
isolated copy of the repository (services/api without .venv, packages, poc); then
`pytest -x -q -p no:cacheprovider tests/engine tests/protocol` runs in the copy with the
project's interpreter. A mutant is KILLED when the suite fails. The unmutated copy is the
control and must pass."""
import shutil
import subprocess
import sys
import tarfile
from concurrent.futures import ThreadPoolExecutor
from pathlib import Path

REPO = Path("/projects/cardflow")
WORK = Path("/tmp/t004c/mut")
PY = REPO / "services/api/.venv/bin/python"
MUTANTS = [
    ("CONTROL-unmutated", None, None, None),
    ("M05-repay-before-txn-off-by-one", "allocation.py",
     "    if info.repay_on < req.txn_date:\n",
     "    if info.repay_on <= req.txn_date:\n"),
    ("M14-verifier-fee-rounds-down", "verifier.py",
     'apply_rate(amount, card.fee_rate, "UP")',
     'apply_rate(amount, card.fee_rate, "DOWN")'),
    ("M15-verifier-benefit-rounds-up", "verifier.py",
     'apply_rate(amt, c.verified_benefit_rate, "DOWN")',
     'apply_rate(amt, c.verified_benefit_rate, "UP")'),
    ("M17-model-benefit-ignores-cap", "allocation.py",
     "            cap = c.verified_benefit_cap if c.verified_benefit_cap is not None else BIG\n",
     "            cap = BIG\n"),
    ("M20-stale-boundary", "plan.py",
     "        elif (req.txn_date - s.as_of).days > stale_after_days:\n",
     "        elif (req.txn_date - s.as_of).days >= stale_after_days:\n"),
    ("M26-weighted-buffer-half-even", "verifier.py",
     'quantize(Decimal("0.1"), rounding=ROUND_HALF_UP)',
     'quantize(Decimal("0.1"), rounding="ROUND_HALF_EVEN")'),
]


def copy_repo(dest: Path) -> None:
    if dest.exists():
        shutil.rmtree(dest)
    dest.mkdir(parents=True)
    skip = {".venv", "__pycache__", ".pytest_cache", ".ruff_cache", ".hypothesis"}
    for part in ("services/api", "packages", "poc"):
        src = REPO / part
        shutil.copytree(src, dest / part, ignore=lambda d, names: [n for n in names if n in skip])


def run(m: tuple) -> str:
    name, file, old, new = m
    root = WORK / name
    copy_repo(root)
    if file is not None:
        target = root / "services/api/cardflow/planner/engine" / file
        text = target.read_text()
        assert text.count(old) == 1, (name, "mutation site not found exactly once")
        target.write_text(text.replace(old, new))
    api = root / "services/api"
    proc = subprocess.run(
        [str(PY), "-m", "pytest", "-x", "-q", "-p", "no:cacheprovider", "tests/engine", "tests/protocol"],
        cwd=api, capture_output=True, text=True, env={"PYTHONPATH": str(api), "PATH": "/usr/bin:/bin"},
    )
    lines = proc.stdout.splitlines()
    failed = [l for l in lines if l.startswith("FAILED")]
    summary = lines[-1] if lines else proc.stderr[-300:]
    verdict = "PASSES (control)" if file is None and proc.returncode == 0 else (
        "KILLED" if proc.returncode != 0 else "SURVIVED")
    site = "" if file is None else f"\n   {file}: {old.strip()!r}\n        -> {new.strip()!r}"
    return f"{name}: {verdict} (pytest exit {proc.returncode}){site}\n   {summary}\n   " + (
        failed[0] if failed else "(no failing test)")


with ThreadPoolExecutor(max_workers=4) as pool:
    results = list(pool.map(run, MUTANTS))
print("\n".join(results))
killed = sum(r.split(": ")[1].startswith("KILLED") for r in results[1:])
control = results[0].split(": ")[1].startswith("PASSES")
print(f"\nkilled {killed}/{len(MUTANTS) - 1}; control passes: {control}")
sys.exit(0 if killed == len(MUTANTS) - 1 and control else 1)
```
