import json
import sys
from pathlib import Path

from cardflow.planner.engine.protocol import parse_json, plan_from_protocol

I63 = 2**63 - 1
repo = Path(sys.argv[1])

base = parse_json(fixture.read_bytes())["input"]
CASES = [
    ("cards[0].raw_available_minor = 2^63-1", lambda d: d["cards"][0].update(raw_available_minor=I63)),
    ("request.amount_minor = 2^62", lambda d: d["request"].update(amount_minor=2**62)),
    ("request.amount_minor = 2^63-1", lambda d: d["request"].update(amount_minor=I63)),
    ("snapshots[0].balance_minor = -2^63", lambda d: d["snapshots"][0].update(balance_minor=-(2**63))),
    ("snapshots[0].balance_minor = 2^63-1", lambda d: d["snapshots"][0].update(balance_minor=I63)),
    ("events[0].amount_minor = 2^63-1", lambda d: d["events"][0].update(amount_minor=I63)),
    ("snapshots[0].balance_minor = 10^13", lambda d: d["snapshots"][0].update(balance_minor=10**13)),
    ("cards[0].fee_rate = '0.006'", lambda d: d["cards"][0].update(fee_rate="0.006")),
    ("cards[0].fee_rate = '0.1234567891'", lambda d: d["cards"][0].update(fee_rate="0.1234567891")),
    ("cards[0].fee_rate = '9999999999.9999999999'", lambda d: d["cards"][0].update(fee_rate="9999999999.9999999999")),
    ("request.payment_lead_days = 60", lambda d: d["request"].update(payment_lead_days=60)),
]
for name, mutate in CASES:
    doc = json.loads(json.dumps(base))
    mutate(doc)
    try:
        out = plan_from_protocol(doc)
        alloc = {a["card_id"]: a["amount_minor"] for a in out["allocation"]}
        note = out["notes"][-1][:90] if out["notes"] else ""
        sys.stdout.write(f"{name}: {out['business_status']} / {out['solver_status']} {alloc} {note!r}\n")
    except Exception as e:  # noqa: BLE001 -- the probe reports every exception type
        sys.stdout.write(f"{name}: EXCEPTION {type(e).__name__}: {str(e).splitlines()[0][:120]}\n")
