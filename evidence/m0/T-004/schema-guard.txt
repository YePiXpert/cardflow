"""T-004 schema guard inventory (one-off): runs the guard functions of
services/api/tests/protocol/test_schemas.py on the committed schema files and lists what they
look at. usage: python schema_inventory.py REPO_ROOT"""

import importlib.util
import json
import sys
from pathlib import Path

from jsonschema import Draft202012Validator

repo = Path(sys.argv[1])
spec = importlib.util.spec_from_file_location(
    "t004_test_schemas", repo / "services/api/tests/protocol/test_schemas.py"
)
guards = importlib.util.module_from_spec(spec)
spec.loader.exec_module(guards)
schema_dir = repo / "packages/rules-protocol/schema/v1"
schemas = {p.name: json.loads(p.read_text(encoding="utf-8")) for p in sorted(schema_dir.glob("*.json"))}

print("## meta-schema (Draft 2020-12) check per file")
for name, schema in schemas.items():
    Draft202012Validator.check_schema(schema)
    print(f"  OK  {name}  $id={schema['$id']}")
print("\n## 'number' type occurrences (whitelist: " + ", ".join(sorted(f"{f}:{p}" for f, p in guards.NUMBER_WHITELIST)) + ")")
for occurrence in guards.number_occurrences(schemas):
    print(f"  {occurrence}")
print(f"  violations: {guards.number_violations(schemas)}")
print("\n## money members (property -> referenced definition)")
money = rates = closed = maps = 0
for file, pointer, node in guards.walk(schemas):
    name = guards._property_name(pointer)
    if name is not None and (name.endswith("_minor") or name in guards.OUTPUT_MONEY):
        target = guards._ref_target(node) or (
            "map of " + str(guards._ref_target(node.get("additionalProperties", {})))
        )
        print(f"  {file}#{pointer} -> {target}")
        money += 1
    if name is not None and ("rate" in name or name.endswith("_scale")):
        rates += 1
        print(f"  (rate) {file}#{pointer} -> {guards._ref_target(node)}")
    if "object" in guards._types(node) or isinstance(node.get("properties"), dict):
        if node.get("additionalProperties") is False:
            closed += 1
        else:
            maps += 1
print(f"  money members: {money}; rate/factor members: {rates}; violations: {guards.money_violations(schemas)}")
common = schemas["common.schema.json"]["$defs"]
for d in sorted(guards.MONEY_DEFS):
    print(f"  common $defs/{d}: {json.dumps({k: v for k, v in common[d].items() if k != 'description'})}")
print(f"\n## object schemas: {closed} closed (additionalProperties: false), {maps} id-keyed maps; violations: {guards.open_object_violations(schemas)}")
