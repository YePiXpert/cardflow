```python
"""T-004 B.6: which top-level definitions of a module changed beyond annotations (one-off tool).

Both files are parsed; every annotation is removed (argument/return annotations, annotated
assignments keep target and value), docstrings are dropped, then each top-level function/class
is compared with ast.dump. usage: python ast_annotations.py OLD NEW
"""

import ast
import sys
from pathlib import Path


class Strip(ast.NodeTransformer):
    def _args(self, a: ast.arguments) -> None:
        for arg in [*a.posonlyargs, *a.args, *a.kwonlyargs, a.vararg, a.kwarg]:
            if arg is not None:
                arg.annotation = None

    def visit_FunctionDef(self, node: ast.FunctionDef) -> ast.AST:
        node.returns = None
        self._args(node.args)
        if node.body and isinstance(node.body[0], ast.Expr) and isinstance(node.body[0].value, ast.Constant):
            node.body = node.body[1:] or [ast.Pass()]
        return self.generic_visit(node)

    def visit_AnnAssign(self, node: ast.AnnAssign) -> ast.AST:
        if node.value is None:
            return node
        return ast.copy_location(ast.Assign(targets=[node.target], value=node.value), node)


def defs(path: str) -> dict[str, str]:
    tree = Strip().visit(ast.parse(Path(path).read_text(encoding="utf-8")))
    return {
        n.name: ast.dump(n)
        for n in tree.body
        if isinstance(n, (ast.FunctionDef, ast.ClassDef))
    }


old, new = defs(sys.argv[1]), defs(sys.argv[2])
for name in sorted(set(old) | set(new)):
    if name not in old:
        state = "ADDED"
    elif name not in new:
        state = "REMOVED"
    elif old[name] == new[name]:
        state = "identical after stripping annotations and docstrings"
    else:
        state = "DIFFERS (behaviour or signature changed)"
    print(f"{name}: {state}")
```
