    """Replays test_signal_handler_takes_no_lock with the OLD handler (Event.set in the handler)."""
    import faulthandler
    import os
    import signal
    import sys
    import time
    
    sys.path.insert(0, "/projects/cardflow/services/api/tests/jobs")
    import jobs_testkit
    from jobs_helpers import make_config
    from sqlalchemy.orm import sessionmaker
    
    from cardflow.jobs.worker import Worker
    
    worker = Worker(sessionmaker(), jobs_testkit.build_registry(), make_config())
    
    
    def old_handle(signum, frame):  # what the handler did before the fix
        worker._signal = signum
        worker.request_stop()
    
    
    signal.signal(signal.SIGTERM, old_handle)
    faulthandler.dump_traceback_later(5, exit=True)
    events = (worker._wake, worker._stop, worker._force)
    for event in events:
        event._cond.acquire()
    os.kill(os.getpid(), signal.SIGTERM)
    time.sleep(0.05)
    for event in events:
        event._cond.release()
    print("NO DEADLOCK (unexpected)")
