# 比较脚本（/tmp/evidence-tools/compare_search.py）
"""比较 Dart 有限搜索（K <= 3，--json 输出）与 Python CP-SAT 引擎（plan_from_protocol，不算敏感性）。

对每份随机文档：NEEDS_DATA ↔ needsData（数据问题代码相同）、BASELINE_SHORTFALL ↔ baselineShortfall、
INFEASIBLE ↔ notFound、FEASIBLE_ROBUST/CONDITIONALLY_FEASIBLE ↔ found 且分配、未通过情景、各目标层取值、
验证结果（协议形式）与 CP-SAT 的完全相同。请求的 K > 3 的文档不比较求解结果（本地搜索只到 3 张卡）。
"""
import collections
import json
import sys
from pathlib import Path

from cardflow.planner.engine.protocol import parse_json, plan_from_protocol

dart = [json.loads(line) for line in Path(sys.argv[1]).read_text().splitlines()]
stats = collections.Counter()
mismatches = []
for r in dart:
    doc = parse_json(Path(r["file"]).read_bytes())["input"]
    req = doc["request"]
    k = req["max_cards"] if req["allow_split"] else 1
    out = plan_from_protocol(doc, with_sensitivity=False)
    py, bs = out["business_status"], r["outcome"]
    stats[f"{py} -> {bs}"] += 1
    expected = {
        "NEEDS_DATA": "needsData",
        "BASELINE_SHORTFALL": "baselineShortfall",
        "INFEASIBLE": "notFound",
        "FEASIBLE_ROBUST": "found",
        "CONDITIONALLY_FEASIBLE": "found",
    }.get(py)
    if k > 3 and py in ("FEASIBLE_ROBUST", "CONDITIONALLY_FEASIBLE", "INFEASIBLE"):
        stats["skipped: request K > 3"] += 1
        continue
    if expected != bs:
        mismatches.append((r["file"], py, bs))
        continue
    if bs == "needsData" and [i["code"] for i in out["data_issues"]] != [
        i["code"] for i in r["data_issues"]
    ]:
        mismatches.append((r["file"], "data_issues"))
    if bs in ("found", "notFound") and not r["exhausted"]:
        stats["search not exhausted"] += 1
    if bs == "found":
        pa = [(a["card_id"], a["amount_minor"]) for a in out["allocation"]]
        da = [(a["card_id"], a["amount_minor"]) for a in r["allocation"]]
        stats["found: same allocation as CP-SAT" if pa == da else "found: other allocation"] += 1
        if pa != da:
            mismatches.append((r["file"], "allocation", pa, da))
        if out["failing_scenarios"] != r["failing_scenarios"]:
            mismatches.append((r["file"], "failing_scenarios"))
        for t in out["tiers"]:
            if t["status"] != "SKIPPED_CONSTANT" and r["objectives"][t["name"]] != t["value"]:
                mismatches.append((r["file"], "tier " + t["name"], t["value"], r["objectives"][t["name"]]))
        if out["verification"] != r["verification"]:
            mismatches.append((r["file"], "verification"))
print(json.dumps(dict(sorted(stats.items())), ensure_ascii=False, indent=1))
print("mismatches:", len(mismatches))
for m in mismatches[:20]:
    print(m)
