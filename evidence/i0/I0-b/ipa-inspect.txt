def thin_slices(b):
    magic = struct.unpack('>I', b[:4])[0]
    if magic in (0xcafebabe, 0xcafebabf):
        n = struct.unpack('>I', b[4:8])[0]
        out = []
        for i in range(n):
            if magic == 0xcafebabe:
                cputype, _, off, size, _ = struct.unpack('>iiIII', b[8 + i * 20: 28 + i * 20])
            else:
                cputype, _, off, size, _, _ = struct.unpack('>iiQQII', b[8 + i * 32: 40 + i * 32])
            out.append(b[off:off + size])
        return out
    return [b]


def macho(b):
    """Return (cputype, signature description, id_dylib, load_dylibs)."""
    magic = struct.unpack('<I', b[:4])[0]
    if magic not in (0xfeedfacf, 0xfeedface):
        return None
    is64 = magic == 0xfeedfacf
    cputype, _, _, ncmds, _, _ = struct.unpack('<iiIIII', b[4:28])
    off = 32 if is64 else 28
    sig = 'no LC_CODE_SIGNATURE (unsigned)'
    id_dylib = None
    loads = []
    for _ in range(ncmds):
        cmd, cmdsize = struct.unpack('<II', b[off:off + 8])
        if cmd in (LC_ID_DYLIB, LC_LOAD_DYLIB):
            name_off = struct.unpack('<I', b[off + 8:off + 12])[0]
            raw = b[off + name_off:off + cmdsize]
            name = raw[:raw.index(b'\0')].decode()
            if cmd == LC_ID_DYLIB:
                id_dylib = name
            else:
                loads.append(name)
        if cmd == LC_CODE_SIGNATURE:
            dataoff, datasize = struct.unpack('<II', b[off + 8:off + 16])
            blob = b[dataoff:dataoff + datasize]
            sb_magic, _, count = struct.unpack('>III', blob[:12])
            has_cms = False
            flags = team = None
            for i in range(count):
                _, boff = struct.unpack('>II', blob[12 + i * 8: 20 + i * 8])
                bmagic, blen = struct.unpack('>II', blob[boff:boff + 8])
                if bmagic == 0xfade0b01 and blen > 8:
                    has_cms = True
                if bmagic == 0xfade0c02:
                    cd = blob[boff:boff + blen]
                    version, flags = struct.unpack('>II', cd[8:16])
                    if version >= 0x20200:
                        team_off = struct.unpack('>I', cd[48:52])[0]
                        if team_off:
                            team = cd[team_off:cd.index(b'\0', team_off)].decode()
            adhoc = bool(flags is not None and flags & 0x2)
            sig = f'signed: adhoc={adhoc} cms_signature={has_cms} team_id={team!r}'
        off += cmdsize
    return cputype, sig, id_dylib, loads
