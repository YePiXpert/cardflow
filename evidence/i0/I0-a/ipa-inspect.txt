"""Inspect an unsigned IPA without extracting it: structure, Info.plist, signing leftovers, hashes."""
import hashlib
import io
import plistlib
import sys
import zipfile

path = sys.argv[1]
data = open(path, 'rb').read()
print(f'file: {path.rsplit("/", 1)[-1]}')
print(f'size_bytes: {len(data)}')
print(f'sha256: {hashlib.sha256(data).hexdigest()}')

zf = zipfile.ZipFile(io.BytesIO(data))
names = zf.namelist()
print(f'entries: {len(names)}')
top = sorted({n.split('/')[0] for n in names})
print(f'top_level: {top}')
apps = sorted({'Payload/' + n.split('/')[1] for n in names if n.startswith('Payload/') and n.split('/')[1]})
print(f'payload_children: {apps}')

info = plistlib.loads(zf.read('Payload/Runner.app/Info.plist'))
keys = ['CFBundleIdentifier', 'CFBundleDisplayName', 'CFBundleName', 'CFBundleShortVersionString',
        'CFBundleVersion', 'MinimumOSVersion', 'CFBundleLocalizations', 'CFBundleExecutable',
        'UIDeviceFamily', 'DTPlatformName', 'DTPlatformVersion', 'DTSDKName', 'DTXcode', 'DTXcodeBuild']
for k in keys:
    print(f'Info.plist {k}: {info.get(k)!r}')
print('Info.plist usage descriptions:', sorted(k for k in info if k.endswith('UsageDescription')))
print('Info.plist UIBackgroundModes:', info.get('UIBackgroundModes'))

def exists(suffix):
    return any(n == suffix or n.startswith(suffix + '/') for n in names)

print('embedded.mobileprovision present:', exists('Payload/Runner.app/embedded.mobileprovision'))
print('Runner.app/_CodeSignature present:', exists('Payload/Runner.app/_CodeSignature'))
print('entitlement files:', [n for n in names if n.endswith(('.entitlements', '.xcent'))])
print('nested _CodeSignature dirs:', sorted({n.split('/_CodeSignature/')[0] for n in names if '/_CodeSignature/' in n}))
frameworks = sorted({n.split('/')[3] for n in names if n.startswith('Payload/Runner.app/Frameworks/') and n.count('/') >= 4})
print('Frameworks:', frameworks)
privacy = 'Payload/Runner.app/PrivacyInfo.xcprivacy'
if privacy in names:
    p = plistlib.loads(zf.read(privacy))
    print('PrivacyInfo NSPrivacyTracking:', p.get('NSPrivacyTracking'))
    print('PrivacyInfo APIs:', [(a['NSPrivacyAccessedAPIType'], a['NSPrivacyAccessedAPITypeReasons']) for a in p.get('NSPrivacyAccessedAPITypes', [])])
else:
    print('PrivacyInfo.xcprivacy: MISSING')
checks = {
    'structure Payload/Runner.app only': top == ['Payload'] and apps == ['Payload/Runner.app'],
    'CFBundleIdentifier == li.yep.cardflow': info.get('CFBundleIdentifier') == 'li.yep.cardflow',
    'MinimumOSVersion == 15.0': info.get('MinimumOSVersion') == '15.0',
    'no embedded.mobileprovision': not exists('Payload/Runner.app/embedded.mobileprovision'),
    'PrivacyInfo.xcprivacy bundled': privacy in names,
    'CFBundleDisplayName == 卡序': info.get('CFBundleDisplayName') == '卡序',
}
for k, v in checks.items():
    print(f'CHECK {k}: {"PASS" if v else "FAIL"}')
sys.exit(0 if all(checks.values()) else 1)

"""Report the code-signature state of Mach-O binaries inside an IPA (read in memory, no extraction)."""
import io, struct, sys, zipfile

LC_CODE_SIGNATURE = 0x1d

def thin_slices(b):
    magic = struct.unpack('>I', b[:4])[0]
    if magic in (0xcafebabe, 0xcafebabf):  # fat
        n = struct.unpack('>I', b[4:8])[0]
        out = []
        for i in range(n):
            if magic == 0xcafebabe:
                cputype, _, off, size, _ = struct.unpack('>iiIII', b[8 + i * 20: 28 + i * 20])
            else:
                cputype, _, off, size, _, _ = struct.unpack('>iiQQII', b[8 + i * 32: 40 + i * 32])
            out.append((cputype, b[off:off + size]))
        return out
    return [(None, b)]

def describe(b):
    magic = struct.unpack('<I', b[:4])[0]
    if magic not in (0xfeedfacf, 0xfeedface):
        return 'not a Mach-O'
    is64 = magic == 0xfeedfacf
    cputype, _, _, ncmds, _, _ = struct.unpack('<iiIIII', b[4:28])
    off = 32 if is64 else 28
    for _ in range(ncmds):
        cmd, cmdsize = struct.unpack('<II', b[off:off + 8])
        if cmd == LC_CODE_SIGNATURE:
            dataoff, datasize = struct.unpack('<II', b[off + 8:off + 16])
            blob = b[dataoff:dataoff + datasize]
            sb_magic, _, count = struct.unpack('>III', blob[:12])
            if sb_magic != 0xfade0cc0:
                return f'cpu={cputype} LC_CODE_SIGNATURE present, unknown superblob'
            has_cms = False
            flags = team = None
            for i in range(count):
                _, boff = struct.unpack('>II', blob[12 + i * 8: 20 + i * 8])
                bmagic, blen = struct.unpack('>II', blob[boff:boff + 8])
                if bmagic == 0xfade0b01 and blen > 8:
                    has_cms = True
                if bmagic == 0xfade0c02:
                    cd = blob[boff:boff + blen]
                    version, flags = struct.unpack('>II', cd[8:16])
                    if version >= 0x20200:
                        team_off = struct.unpack('>I', cd[48:52])[0]
                        if team_off:
                            team = cd[team_off:cd.index(b'\0', team_off)].decode()
            adhoc = bool(flags is not None and flags & 0x2)
            return f'cpu={cputype} signed: adhoc={adhoc} cms_signature={has_cms} team_id={team!r}'
        off += cmdsize
    return f'cpu={cputype} no LC_CODE_SIGNATURE (unsigned)'

zf = zipfile.ZipFile(io.BytesIO(open(sys.argv[1], 'rb').read()))
for name in ('Payload/Runner.app/Runner',
             'Payload/Runner.app/Frameworks/App.framework/App',
             'Payload/Runner.app/Frameworks/Flutter.framework/Flutter'):
    data = zf.read(name)
    for cpu, sl in thin_slices(data):
        print(f'{name}: {describe(sl)}')
