# 分期拆分与成本的 Python 对照样本（services/api/cardflow/planner/engine/money.py，只读调用）。
# 每个样本输出一行 JSON：拆分、各期 (期号, 本金, 费用)、月 IRR（放款为负、各期还款为正）、
# annualize(每期费率) 与 annualize(月 IRR)。Decimal 一律以 str() 输出。
import json
import random
from decimal import Decimal

from cardflow.planner.engine.money import (
    annualize,
    installment_cash_flows,
    irr_monthly,
    split_installments,
)

samples = [
    ("F-36", 1000000, 12, "0.006", "last"),
    ("F-37", 100000, 12, "0.015", "last"),
    ("remainder_first", 1000000, 7, "0.0075", "first"),
    ("single_period", 12345, 1, "0.02", "last"),
]
rng = random.Random(20261019)
for i in range(3):
    principal = rng.randint(1000, 10**9)
    periods = rng.randint(2, 36)
    rate = str(Decimal(rng.randint(1, 200)) / Decimal(10000))
    remainder = rng.choice(["last", "first"])
    samples.append((f"random_{i + 1}", principal, periods, rate, remainder))

for name, principal, periods, rate, remainder in samples:
    split = split_installments(principal, periods, remainder)
    flows = installment_cash_flows(principal, periods, rate, remainder)
    irr = irr_monthly([(0, -principal)] + [(k, p + f) for k, p, f in flows])
    a_rate = annualize(Decimal(rate))
    a_irr = annualize(irr)
    print(json.dumps({
        "name": name,
        "principal": principal,
        "periods": periods,
        "rate": rate,
        "remainder": remainder,
        "split": split,
        "flows": [list(f) for f in flows],
        "irr": str(irr),
        "rate_nominal_x12": str(a_rate["nominal_x12"]),
        "rate_effective_compound": str(a_rate["effective_compound"]),
        "irr_nominal_x12": str(a_irr["nominal_x12"]),
        "irr_effective_compound": str(a_irr["effective_compound"]),
    }, ensure_ascii=False))
