#       另附本机对两次 IPA 的只读核对（python3 -I，内存中解析，不解压）。过滤：无。

# 结论（规则：主二进制或没有自带清单的 framework 命中的类别保留在 App 清单，其余删除，理由码不变）：
# - UserDefaults（CA92.1）：主二进制、App/objective_c/sqlite3mc/Flutter 都没有 _OBJC_CLASS_$_NSUserDefaults 或 NSUserDefaults 字样 → 删除（8149c29）。
# - FileTimestamp（C617.1）：主二进制导入 _NSFileCreationDate（二进制里有 installTime/updateTime 字样，应是 package_info_plus 读安装时间），
#   sqlite3mc（没有自带清单）导入 _stat/_fstat/_lstat → 保留。
# - SystemBootTime（35F9.1）：sqlite3mc 导入 _sysctlbyname，扫描规则算命中 → 保留。补充核对：sqlite3mc 里唯一的 sysctl 名称字符串是 hw.ncpu，
#   没有 kern.boottime、systemUptime、mach_absolute_time——它不是在读开机时间，这一条属多报；按扫描规则与"多报不影响安装"（BL-16）保守保留，
#   是否删除请主会话裁决（删除要再构建一次）。
# - DiskSpace（E174.1）：sqlite3mc 导入 _statfs/_fstatfs → 保留（用途未逐行核实，按规则保留）。
# - 自带清单：Flutter.framework 有（FileTimestamp 0A2A.1、C617.1；SystemBootTime 35F9.1），它的命中由它自己的清单承担；App.framework、
#   objective_c.framework、sqlite3mc.framework 没有；资源包 flutter_secure_storage_darwin、package_info_plus 自带清单（API 列表为空）。
# - PrivacyInfo 有改动：8149c29 删除 UserDefaults，保留其余三类（理由码不变）；交付的是第 2 次构建的 IPA。第 2 次扫描与第 1 次的命中相同，
#   App 清单只剩三类，结论一致（UserDefaults 未登记、仍无命中）。

## 本机补充核对（只读）：sysctl 名称字符串与相关字样
$ cat sysctl_names.py
"""对 IPA 里的二进制做只读核对（内存中解析，不解压）：sysctl 名称字符串、开机时间与磁盘空间相关字样、
主二进制里 NSFileCreationDate 的上下文字样（哪个插件在读文件时间戳）。用法：sysctl_names.py IPA"""
import re
import sys
import zipfile

zf = zipfile.ZipFile(sys.argv[1])
binaries = [
    'Payload/Runner.app/Runner',
    'Payload/Runner.app/Frameworks/App.framework/App',
    'Payload/Runner.app/Frameworks/objective_c.framework/objective_c',
    'Payload/Runner.app/Frameworks/sqlite3mc.framework/sqlite3mc',
]
for name in binaries:
    b = zf.read(name)
    names = sorted({m.group(0).decode() for m in re.finditer(
        rb'(?<![A-Za-z0-9_.])(?:hw|kern|machdep|vm|net|sysctl)\.[A-Za-z0-9_.]{2,60}', b)})
    print(f'{name[len("Payload/Runner.app/"):]}: sysctl 名称字符串 = {names}')
    for kw in (b'kern.boottime', b'systemUptime', b'mach_absolute_time', b'statfs', b'NSUserDefaults',
               b'NSFileCreationDate', b'installTime', b'updateTime'):
        print(f'  含 {kw.decode()!r}: {kw in b}')
