def png_chunks(png):
    """[(chunk type, data)]；Xcode 的 actool 产出的图标是 Apple 的 CgBI 变体（IHDR 之前多一个 CgBI 块）。"""
    off, out = 8, []
    while off + 8 <= len(png):
        length, kind = struct.unpack('>I4s', png[off:off + 8])
        out.append((kind.decode('latin-1'), png[off + 8:off + 8 + length]))
        off += 12 + length
    return out


def png_info(png):

    names = [k for k, _ in chunks]
    return {'w': w, 'h': h, 'depth': depth, 'color': color, 'chunks': names}

def png_pixels(png):
    """解码 8 位 RGB/RGBA PNG 与 Apple 的 CgBI 变体（IDAT 为无 zlib 头的 deflate，像素 BGRA、预乘 alpha），
    返回 (宽, 高, [(r, g, b, a), ...])。只支持本工程图标用到的格式（8 位、颜色类型 2 或 6、无隔行）。"""
    chunks = png_chunks(png)
    names = [k for k, _ in chunks]
    w, h, depth, color, _, _, interlace = struct.unpack('>IIBBBBB', dict(chunks)['IHDR'][:13])
    assert depth == 8 and color in (2, 6) and interlace == 0, (depth, color, interlace)
    data = b''.join(d for k, d in chunks if k == 'IDAT')
    raw = zlib.decompressobj(-15).decompress(data) if 'CgBI' in names else zlib.decompress(data)
    bpp = 4 if color == 6 else 3
    stride = w * bpp
    rows, prev, off = [], bytearray(stride), 0
    for _ in range(h):
        ftype, line = raw[off], bytearray(raw[off + 1:off + 1 + stride])
        off += 1 + stride
        for i in range(stride):
            a = line[i - bpp] if i >= bpp else 0
            b = prev[i]
            c = prev[i - bpp] if i >= bpp else 0
            if ftype == 1:
                line[i] = (line[i] + a) & 0xFF
            elif ftype == 2:
                line[i] = (line[i] + b) & 0xFF
            elif ftype == 3:
                line[i] = (line[i] + (a + b) // 2) & 0xFF
            elif ftype == 4:
                pa, pb, pc = abs(b - c), abs(a - c), abs(a + b - 2 * c)
                line[i] = (line[i] + (a if pa <= pb and pa <= pc else b if pb <= pc else c)) & 0xFF
        rows.append(bytes(line))
        prev = line
    pixels = []
    for line in rows:
        for x in range(w):
            px = line[x * bpp:(x + 1) * bpp]
            if 'CgBI' in names:
                bl, g, r, a = px
            elif bpp == 4:
                r, g, bl, a = px
            else:
                r, g, bl, a = px[0], px[1], px[2], 255
            pixels.append((r, g, bl, a))
    return w, h, pixels

def thin_slices(b):
    magic = struct.unpack('>I', b[:4])[0]
    if magic in (0xcafebabe, 0xcafebabf):
        n = struct.unpack('>I', b[4:8])[0]
        out = []
        for i in range(n):
            if magic == 0xcafebabe:
                cputype, _, off, size, _ = struct.unpack('>iiIII', b[8 + i * 20: 28 + i * 20])
            else:
                cputype, _, off, size, _, _ = struct.unpack('>iiQQII', b[8 + i * 32: 40 + i * 32])
            out.append(b[off:off + size])
        return out
    return [b]


def macho(b):
    """Return (cputype, signature description, id_dylib, load_dylibs)."""
    magic = struct.unpack('<I', b[:4])[0]
    if magic not in (0xfeedfacf, 0xfeedface):
        return None
    is64 = magic == 0xfeedfacf
    cputype, _, _, ncmds, _, _ = struct.unpack('<iiIIII', b[4:28])
    off = 32 if is64 else 28
    sig = 'no LC_CODE_SIGNATURE (unsigned)'
    id_dylib = None
    loads = []
    for _ in range(ncmds):
        cmd, cmdsize = struct.unpack('<II', b[off:off + 8])
        if cmd in (LC_ID_DYLIB, LC_LOAD_DYLIB):
            name_off = struct.unpack('<I', b[off + 8:off + 12])[0]
            raw = b[off + name_off:off + cmdsize]
            name = raw[:raw.index(b'\0')].decode()
            if cmd == LC_ID_DYLIB:
                id_dylib = name
            else:
                loads.append(name)
        if cmd == LC_CODE_SIGNATURE:
            dataoff, datasize = struct.unpack('<II', b[off + 8:off + 16])
            blob = b[dataoff:dataoff + datasize]
            sb_magic, _, count = struct.unpack('>III', blob[:12])
            has_cms = False
            flags = team = None
            for i in range(count):
                _, boff = struct.unpack('>II', blob[12 + i * 8: 20 + i * 8])
                bmagic, blen = struct.unpack('>II', blob[boff:boff + 8])
                if bmagic == 0xfade0b01 and blen > 8:
                    has_cms = True
                if bmagic == 0xfade0c02:
                    cd = blob[boff:boff + blen]
                    version, flags = struct.unpack('>II', cd[8:16])
                    if version >= 0x20200:
                        team_off = struct.unpack('>I', cd[48:52])[0]
                        if team_off:
                            team = cd[team_off:cd.index(b'\0', team_off)].decode()
            adhoc = bool(flags is not None and flags & 0x2)
            sig = f'signed: adhoc={adhoc} cms_signature={has_cms} team_id={team!r}'
        off += cmdsize
    return cputype, sig, id_dylib, loads
